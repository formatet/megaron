 "context"

 "encoding/json"

 "os"
 "os/exec"

 "testing"

 "formatet/megaron/server/internal/auth"
 "github.com/go-chi/chi/v5"
 "github.com/google/uuid"

// Isolated synthetic cities; actual authentication, read/write handlers, CLI and browser.
func TestRelationPlayerScenario(t *testing.T) {
 root,err:=filepath.Abs("../../..");if err!=nil {t.Fatal(err)}
 cli:=filepath.Join(t.TempDir(),"keryx")
 build:=exec.Command("go","build","-o",cli,"./cmd/keryx");build.Dir=filepath.Join(root,"server")
 if out,err:=build.CombinedOutput();err!=nil {t.Fatalf("build: %v %s",err,out)}
 for run:=1;run<=2;run++ {
  f:=newAlphaFixture(t);relationValues(t,f);var prov uuid.UUID
  if err:=f.pool.QueryRow(context.Background(),`SELECT province_id FROM settlements WHERE id=$1`,f.cities[0]).Scan(&prov);err!=nil {t.Fatal(err)}
  before:=relationTuple(t,f)
  r:=chi.NewRouter()
  r.Handle("/static/*",http.StripPrefix("/static/",http.FileServer(http.Dir(filepath.Join(root,"web/static")))))
  r.Get("/fixture",func(w http.ResponseWriter,req *http.Request){http.ServeFile(w,req,filepath.Join(root,"web/static/map.html"))})
  ph:=NewProvinceHandler(f.pool,nil,f.h.clk,nil,nil)
  r.Group(func(r chi.Router){r.Use(auth.Middleware(auth.NewService(f.pool,"alpha-secret")))
   r.Get("/api/v1/worlds/{worldID}/provinces/{provinceID}",ph.Get)
   r.Get("/api/v1/worlds/{worldID}/settlements/{settlementID}",f.h.Get)
   r.Post("/api/v1/worlds/{worldID}/settlements/{settlementID}/rite",f.h.Rite)
   r.Get("/api/v1/worlds/{worldID}/provinces",func(w http.ResponseWriter,req *http.Request){json.NewEncoder(w).Encode([]map[string]any{{"id":prov,"settlement_id":f.cities[0],"own":true,"is_capital":true,"name":"Alpha City","q":0,"r":0}})})
   r.Get("/api/v1/worlds/{worldID}/provinces/{provinceID}/actions",func(w http.ResponseWriter,req *http.Request){w.Write([]byte("[]"))})
  })
  server:=httptest.NewServer(r)
  input,_:=json.Marshal(map[string]any{"base":server.URL,"token":f.token,"world":f.world,"player":f.player,"city":f.cities[0],"province":prov,"cli":cli,"run":run})
  cmd:=exec.Command("python3",filepath.Join(root,"docs/reviews/bon-relation/player_scenario.py"));cmd.Env=os.Environ();cmd.Stdin=bytes.NewReader(input)
  out,err:=cmd.CombinedOutput();server.Close();if err!=nil {t.Fatalf("scenario run%d: %v %s",run,err,out)};t.Logf("scenario run%d: %s",run,out)
  if relationTuple(t,f)!=before {t.Fatal("client scenario changed standing")}
  if f.stock(t,f.cities[0],"grain")!=92 || f.stock(t,f.cities[0],"timber")!=92 {t.Fatal("client scenario goods debit mismatch")}
 }
}
